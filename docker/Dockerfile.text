# Dockerfile.text — self-built text server image (vLLM on CUDA).
#
# Use this when you want vLLM's paged-attention memory efficiency instead of the
# prebuilt TGI image. Exposes an OpenAI-compatible API on :8000.
#
#   docker build -f docker/Dockerfile.text -t croft/text:vllm .
#   docker run --gpus all -p 8000:8000 -v $PWD/models:/models \
#       -e MODEL=/models/qwen2.5-7b-q4 croft/text:vllm

FROM nvidia/cuda:12.4.0-runtime-ubuntu22.04@sha256:af8bd179ed3bf69d4b63b19a763662a6141f0f62ef099283f68d0b14b4bab0e3

ENV DEBIAN_FRONTEND=noninteractive \
    PIP_NO_CACHE_DIR=1 \
    HF_HOME=/models/.hf-cache

RUN apt-get update && apt-get install -y --no-install-recommends \
        python3 python3-pip git && \
    rm -rf /var/lib/apt/lists/*

# vLLM pulls in a CUDA-matched torch build.
RUN pip3 install --upgrade pip && \
    pip3 install "vllm==0.6.3"

WORKDIR /app
COPY installer/ /app/installer/
COPY models_manifest.json /app/

EXPOSE 8000
ENV MODEL=mistralai/Mistral-7B-Instruct-v0.2 \
    GPU_MEM_UTIL=0.90 \
    MAX_MODEL_LEN=8192

# Health check hits the models endpoint once the server is warm.
HEALTHCHECK --interval=30s --timeout=5s --start-period=180s --retries=5 \
    CMD python3 -c "import urllib.request;urllib.request.urlopen('http://localhost:8000/v1/models')" || exit 1

CMD python3 -m vllm.entrypoints.openai.api_server \
    --model "$MODEL" --port 8000 \
    --gpu-memory-utilization "$GPU_MEM_UTIL" \
    --max-model-len "$MAX_MODEL_LEN"
